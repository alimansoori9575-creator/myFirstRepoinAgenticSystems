-- 4. Retrieve users who have never placed an order.(Hint: Use LEFT JOIN and check for NULL values.)
SELECT users.name
FROM users
LEFT JOIN order_
ON users.user_id = order_.user_id
WHERE order_.user_id IS NULL

-- 5. Combine all three tables to return:
    -user_id
    -name
    -total_order_amount
    -login_count
SELECT users.name,
SUM(order_.amount) AS total_order_amount,
activity.login_count
FROM users
LEFT JOIN order_
ON users.user_id = order_.user_id
LEFT JOIN activity
ON users.user_id = activity.user_id
GROUP BY users.user_id, order_.amount, activity.login_count; 

-- 6. Retrieve users who placed orders greater than 400.
SELECT users.name, order_.amount
FROM users
INNER JOIN order_
ON users.user_id = order_.user_id
WHERE order_.amount > 400;

-- 7. Retrieve the top 3 users by total order amount.
SELECT users.name,
SUM(order_.amount)
AS total_order_amount
FROM users
INNER JOIN order_
ON users.user_id = order_.user_id
GROUP BY users.name
ORDER BY total_order_amount DESC
LIMIT 3;

--8. Retrieve users sorted by:
    -total order amount (descending)
    -login count (descending)
SELECT users.name, 
SUM(order_.amount) AS total_order_amount,
activity.login_count
FROM users
LEFT JOIN order_
ON users.user_id = order_.user_id
LEFT JOIN activity
ON users.user_id = activity.user_id
GROUP BY users.name, activity.login_count
ORDER BY total_order_amount DESC, activity.login_count DESC;

--9. Return the average order amount per user.
SELECT users.name,
AVG(order_.amount) AS avg_order_amount
FROM users
INNER JOIN order_
ON users.user_id = order_.user_id
GROUP BY users.name;

--10. Write one combined query that:
	 -Selects user_id, name, total_order_amount
	 -Includes login counts
	 -Sorts by total order amount (descending)
   -Returns top 3 users.
SELECT users.user_id,
users.name,
SUM(order_.amount) AS  total_order_amount,
activity.login_count
FROM users
INNER JOIN order_
ON users.user_id = order_.user_id
INNER JOIN activity
ON users.user_id = activity.user_id 
GROUP BY users.user_id, users.name, activity.login_count
ORDER BY activity.login_count DESC
